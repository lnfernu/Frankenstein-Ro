		mes "Você quer obter o " + mesitemlink(.@target) + "?";
		mes "Então, o que gostaria de trocar por ele?";
		next;

		if (countitem(.@Drabone_L) >= 1 && countitem(.@Drabone_S) >= .@ossos_demanda) {
			set .@has_ossos, 1;
			set .@opt1$, "Ossos de Dragão";
		} else {
			set .@has_ossos, 0;
			set .@opt1$, "Ossos de Dragão (Itens Insuficientes)";
		}

		set .@esq_g, countitem(.@Dragenergy_Green);
		set .@esq_b, countitem(.@Dragenergy_Blue);
		set .@esq_r, countitem(.@Dragenergy_Red);
		set .@esq_go, countitem(.@Dragenergy_Gold);
		set .@esq_p, countitem(.@Dragenergy_Purple);
		set .@esq_s, countitem(.@Dragenergy_Silver);

		if (.@esq_g >= .@esferas_qtd && .@esq_b >= .@esferas_qtd && .@esq_r >= .@esferas_qtd && .@esq_go >= .@esferas_qtd && .@esq_p >= .@esferas_qtd && .@esq_s >= .@esferas_qtd) {
			set .@has_esferas, 1;
			set .@opt2$, "Esferas de Dragão";
		} else {
			set .@has_esferas, 0;
			set .@opt2$, "Esferas de Dragão (Itens Insuficientes)";
		}

		set .@material_opt, select(.@opt1$ + ":" + .@opt2$ + ":Deixe-me pensar sobre isso.");

		if (.@material_opt == 1) {
			if (.@has_ossos) {
				mes "[Cerúlio]";
				mes "Deixe-me conferir seus materiais.";
				next;

				delitem .@Drabone_L, 1;
				delitem .@Drabone_S, .@ossos_demanda;

				mes "[Cerúlio]";
				mes "Aqui está!";
				getitem .@target, 1;
				next;
				mes "[Cerúlio]";
				mes "Seu " + mesitemlink(.@target) + " está pronto!";
				close;
			} else {
				mes "[Cerúlio]";
				mes "Você não tem todos os materiais necessários.";
				close;
			}
		}
		else if (.@material_opt == 2) {
			if (.@has_esferas) {
				mes "[Cerúlio]";
				mes "Deixe-me conferir seus materiais.";
				next;

				delitem .@Dragenergy_Green, .@esferas_qtd;
				delitem .@Dragenergy_Blue, .@esferas_qtd;
				delitem .@Dragenergy_Red, .@esferas_qtd;
				delitem .@Dragenergy_Gold, .@esferas_qtd;
				delitem .@Dragenergy_Purple, .@esferas_qtd;
				delitem .@Dragenergy_Silver, .@esferas_qtd;

				mes "[Cerúlio]";
				mes "Aqui está!";
				getitem .@target, 1;
				next;
				mes "[Cerúlio]";
				mes "Seu " + mesitemlink(.@target) + " está pronto!";
				close;
			} else {
				mes "[Cerúlio]";
				mes "Você não tem todos os materiais necessários.";
				close;
			}
		}
		else if (.@material_opt == 3) {
			mes "[Cerúlio]";
			mes "Pense bem, o poder dos dragões não é para qualquer um.";
			close;
		}
		close;
	}
	else if (.@menu == 2) {
		mes "[Cerúlio]";
		mes "Que tipo de complemento você gostaria de saber os materiais?";
		next;

		set .@info, select("Turbulência do Dragão:Poder do Dragão:Força do Dragão:Voltar.");

		if (.@info == 1) {
			mes "[Cerúlio]";
			mes "Para fazer a " + mesitemlink(.@Abyss_Ddbox) + ", você pode escolher entre:";
			next;

			mes "[Cerúlio]";
			mes "^FF0000Opção de Ossos:^000000";
			mes "1x " + mesitemlink(.@Drabone_L) + ",";
			mes "5x " + mesitemlink(.@Drabone_S) + ".";
			next;

			mes "[Cerúlio]";
			mes "^FF0000Ou Opção de Esferas:^000000";
			mes "12x " + mesitemlink(.@Dragenergy_Green) + ",";
			mes "12x " + mesitemlink(.@Dragenergy_Blue) + ",";
			mes "12x " + mesitemlink(.@Dragenergy_Red);
			mes "12x " + mesitemlink(.@Dragenergy_Gold) + ",";
			mes "12x " + mesitemlink(.@Dragenergy_Purple) + ",";
			mes "12x " + mesitemlink(.@Dragenergy_Silver);
			close;
		}
		else if (.@info == 2) {
			mes "[Cerúlio]";
			mes "Para fazer o " + mesitemlink(.@Abyss_Ddbox3) + ", você pode escolher entre:";
			next;

			mes "[Cerúlio]";
			mes "^FF0000Opção de Ossos:^000000";
			mes "1x " + mesitemlink(.@Drabone_L) + ",";
			mes "2x " + mesitemlink(.@Drabone_S) + ".";
			next;

			mes "[Cerúlio]";
			mes "^FF0000Ou Opção de Esferas:^000000";
			mes "8x " + mesitemlink(.@Dragenergy_Green) + ",";
			mes "8x " + mesitemlink(.@Dragenergy_Blue) + ",";
			mes "8x " + mesitemlink(.@Dragenergy_Red);
			mes "8x " + mesitemlink(.@Dragenergy_Gold) + ",";
			mes "8x " + mesitemlink(.@Dragenergy_Purple) + ",";
			mes "8x " + mesitemlink(.@Dragenergy_Silver);
			close;
		}
		else if (.@info == 3) {
			mes "[Cerúlio]";
			mes "Para fazer a " + mesitemlink(.@Abyss_Ddbox4) + ", você pode escolher entre:";
			next;

			mes "[Cerúlio]";
			mes "^FF0000Opção de Ossos:^000000";
			mes "1x " + mesitemlink(.@Drabone_L) + ",";
			mes "1x " + mesitemlink(.@Drabone_S) + ".";
			next;

			mes "[Cerúlio]";
			mes "^FF0000Ou Opção de Esferas:^000000";
			mes "7x " + mesitemlink(.@Dragenergy_Green) + ",";
			mes "7x " + mesitemlink(.@Dragenergy_Blue) + ",";
			mes "7x " + mesitemlink(.@Dragenergy_Red);
			mes "7x " + mesitemlink(.@Dragenergy_Gold) + ",";
			mes "7x " + mesitemlink(.@Dragenergy_Purple) + ",";
			mes "7x " + mesitemlink(.@Dragenergy_Silver);
			close;
		}
		close;
	}
	else if (.@menu == 3) {
		mes "[Cerúlio]";
		mes "É possível adicionar mais poder dos dragões aos equipamentos compatíveis!";
		close;
	}
	else if (.@menu == 4) {
		mes "[Cerúlio]";
		mes "Como quiser... mas você vai perder a chance de ter equipamentos ainda mais poderosos.";
		close;
	}
}
